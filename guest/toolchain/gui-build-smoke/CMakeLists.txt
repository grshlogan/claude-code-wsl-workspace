cmake_minimum_required(VERSION 3.14)
project(ClaudeGuiBuildDepsSmoke LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(WIN32)
  enable_language(RC)
  add_executable(windows-gui WIN32 ../mingw64/gui.cpp ../mingw64/gui.rc)
  target_compile_options(windows-gui PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror>")
  target_link_options(windows-gui PRIVATE -municode -static)
  target_link_libraries(windows-gui PRIVATE user32)
else()
  find_package(OpenGL REQUIRED)
  find_package(X11 REQUIRED)
  find_package(CURL REQUIRED)
  find_package(Threads REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  foreach(component X11 Xrandr Xinerama Xcursor Xi Xext)
    if(NOT TARGET X11::${component})
      message(FATAL_ERROR "Missing X11 component: ${component}")
    endif()
  endforeach()
  add_executable(deps-smoke native.cpp)
  target_compile_options(deps-smoke PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(deps-smoke PRIVATE OpenGL::GL CURL::libcurl PkgConfig::GLIB Threads::Threads
    X11::X11 X11::Xrandr X11::Xinerama X11::Xcursor X11::Xi X11::Xext)
endif()
